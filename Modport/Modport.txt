interface bus_if;

    logic data;
    logic valid;
    logic ready;

    modport master (
        output data,
        output valid,
        input  ready
    );

    modport slave (
        input  data,
        input  valid,
        output ready
    );

endinterface


module producer(
    bus_if.master bus
);

    always_comb begin
        bus.data  = 1'b1;   // Ghi vào interface
        bus.valid = 1'b1;   // Ghi vào interface
    end

endmodule


module consumer(
    bus_if.slave bus
);

    always_comb begin
        if (bus.valid)
            bus.ready = 1'b1;   // Ghi vào interface
        else
            bus.ready = 1'b0;
    end

endmodule


module top;

    bus_if bus();

    producer u_producer(bus);
    consumer u_consumer(bus);

endmodule
